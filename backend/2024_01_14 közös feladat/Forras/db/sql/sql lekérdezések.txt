SELECT * from eu_zonak;

SELECT eu_zonak.zonanev, eu_adatok.datum, eu_adatok.szazalekpont FROM eu_zonak 
INNER JOIN eu_zonadatok ON eu_zonak.id = eu_zonadatok.zonakID
INNER JOIN eu_adatok ON eu_zonadatok.adatokID = eu_adatok.id
WHERE eu_zonak.zonanev LIKE "Észak-Európa";
